# Deliberate differences between two `hyprctl binds -j` dumps (before/after the
# cutover), consumed by bind-parity.sh. Combos are `MODMASK KEY [SUBMAP]`:
# Hyprland's numeric modmask (SHIFT 1, CTRL 4, ALT 8, SUPER 64), the key name
# lowercased (or `#N` for a keycode bind), and the submap (empty = global).
# No wildcards: anything not listed here is a violation.

# Bind that exists only after the cutover.
add 64 a                          # SUPER+A  quick settings (remaps.jq shell.quick-settings)
add 76 escape mbm-record          # SUPER+CTRL+ALT+Escape  leaves the key-recording submap (gen-binds.jq)

# Binds that exist only before the cutover. None: every old bind keeps its combo.

# An action may change on a combo present on both sides only if the new action
# matches one of these (ERE against "<dispatcher> <arg>"): the shell IPC remaps.
replace ^exec qs -c mbm ipc call
